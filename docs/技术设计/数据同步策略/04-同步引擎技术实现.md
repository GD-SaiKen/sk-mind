## 3. Connector 工厂

```python
# app/modules/ingestion/connectors/__init__.py

from .base import DatabaseConnector, ApiConnector
from .sqlserver import SqlServerConnector
from .api_client import HttpxApiConnector

def create_database_connector(config: ConnectorConfig) -> DatabaseConnector:
    """根据 config_type 创建对应的数据库连接器"""
    plain_config = decrypt_config(config.encrypted_config)
    if config.config_type == "sqlserver":
        return SqlServerConnector(SqlServerConfig(**plain_config))
    # 后续可扩展 mysql / postgresql
    raise ValueError(f"Unsupported db connector: {config.config_type}")

def create_api_connector(config: ConnectorConfig) -> ApiConnector:
    """创建 API 连接器"""
    plain_config = decrypt_config(config.encrypted_config)
    return HttpxApiConnector(ApiConfig(**plain_config))
```

## 4. RQ 任务设计

### 4.1 任务注册

```python
# app/modules/ingestion/tasks/full_sync.py

from rq import get_current_job
from app.core.database import SessionLocal
from app.modules.ingestion.services.sync_service import SyncService

def run_full_sync(task_id: int, source_object_id: int, batch_id: str):
    """全量同步 RQ 任务入口"""
    job = get_current_job()
    db = SessionLocal()
    try:
        service = SyncService(db)
        service.execute_full_sync(task_id, source_object_id, batch_id, job)
    except Exception as e:
        service.mark_batch_failed(batch_id, str(e))
        raise
    finally:
        db.close()
```

### 4.2 任务入队

```python
# app/modules/ingestion/services/task_service.py

from redis import Redis
from rq import Queue

redis_conn = Redis.from_url(settings.REDIS_URL)
task_queue = Queue("ingestion", connection=redis_conn)

class TaskService:
    def trigger_full_sync(self, task_id: int, source_object_id: int) -> str:
        batch_id = generate_batch_id()
        # 创建批次记录
        batch = IngestionBatch(
            batch_id=batch_id, task_id=task_id,
            status="pending", trigger_type="manual"
        )
        self.db.add(batch)
        self.db.commit()

        # 入队
        job = task_queue.enqueue(
            "app.modules.ingestion.tasks.full_sync.run_full_sync",
            task_id, source_object_id, batch_id,
            job_timeout=3600,  # 1 小时超时
        )
        batch.rq_job_id = job.id
        self.db.commit()
        return batch_id
```

## 5. 同步编排服务

```python
# app/modules/ingestion/services/sync_service.py

class SyncService:
    """同步编排服务"""

    def __init__(self, db: Session):
        self.db = db

    def execute_full_sync(
        self, task_id: int, source_object_id: int,
        batch_id: str, rq_job
    ):
        """执行全量同步（数据库视图模式）"""
        batch = self._get_batch(batch_id)
        source_obj = self.db.get(SourceObject, source_object_id)
        data_source = self.db.get(DataSource, source_obj.data_source_id)
        connector_config = self._get_connector_config(data_source.id)

        # 1. 获取连接器
        connector = create_database_connector(connector_config)
        connector.connect()

        # 2. 更新批次为 running
        self._mark_batch_running(batch, rq_job.id)
        self._update_progress(rq_job, 0, "正在获取表结构...")

        # 3. Schema 检测
        src_columns = connector.fetch_schema(source_obj.object_name)
        self._ensure_raw_table(source_obj.object_name, src_columns)

        # 4. 分批同步
        total = connector.count_rows(source_obj.object_name)
        offset = 0
        batch_size = self._choose_batch_size(total)
        failed = 0

        while offset < total:
            progress = int(offset / total * 100) if total else 100
            self._update_progress(rq_job, progress, f"同步中 {offset}/{total}")

            rows = list(connector.fetch_data(
                source_obj.object_name, offset=offset, limit=batch_size
            ))
            for row in rows:
                try:
                    record = self._build_raw_record(row, batch_id, source_obj)
                    self.db.add(record)
                except Exception as e:
                    failed += 1
                    self._record_error(batch_id, source_obj.object_name, str(e), str(row)[:200])

            self.db.commit()

            # 心跳检测：更新 last_heartbeat
            batch.last_heartbeat = func.now()
            self.db.commit()
            offset += batch_size

        # 5. 完成
        batch.total_input = total
        batch.total_success = total - failed
        batch.total_failed = failed
        batch.status = "success" if failed == 0 else "partial_success"
        batch.finished_at = func.now()
        self.db.commit()

        connector.disconnect()

    def execute_api_pull(
        self, task_id: int, source_object_id: int,
        batch_id: str, rq_job
    ):
        """执行 API 拉取同步（API 模式）"""
        batch = self._get_batch(batch_id)
        source_obj = self.db.get(SourceObject, source_object_id)
        data_source = self.db.get(DataSource, source_obj.data_source_id)
        connector_config = self._get_connector_config(data_source.id)

        # 1. 获取 API 连接器
        api = create_api_connector(connector_config)
        api.connect()
        self._mark_batch_running(batch, rq_job.id)

        # 2. 获取接口配置
        api_cfg = source_obj.api_config
        endpoint = api_cfg["endpoint"]
        method = api_cfg.get("method", "POST")
        page_size = api_cfg.get("default_page_size", 100)
        body_template = api_cfg.get("request_body_template", {})

        # 3. 分页拉取
        total_fetched = 0
        failed = 0
        for page_records in api.fetch_all_pages(
            endpoint, method=method, body=body_template, page_size=page_size
        ):
            try:
                self._write_api_records(
                    page_records, batch_id, data_source.name,
                    endpoint, body_template
                )
                total_fetched += len(page_records)
            except Exception as e:
                failed += len(page_records) if isinstance(page_records, list) else 0
                self._record_error(batch_id, endpoint, str(e))

        # 4. 完成
        batch.total_input = total_fetched
        batch.total_success = total_fetched - failed
        batch.total_failed = failed
        batch.status = "success" if failed == 0 else "partial_success"
        batch.finished_at = func.now()
        self.db.commit()

        api.disconnect()
```

### 7.1 Raw 物理表生成（数据库视图同步）

```python
# app/modules/ingestion/services/schema_service.py

def generate_raw_table_ddl(table_name: str, columns: list[ColumnInfo]) -> str:
    """根据源表 Schema 生成 PostgreSQL Raw 表 DDL"""
    raw_table = f"raw_erp_dbo_{table_name.lower()}"
    col_defs = []
    for col in columns:
        pg_type = map_sqlserver_to_pg(col.data_type)
        nullable = "" if col.is_nullable else " NOT NULL"
        col_defs.append(f"    {col.name} {pg_type}{nullable}")

    # 追加追踪字段
    tracking_cols = [
        "    _raw_id           BIGSERIAL PRIMARY KEY",
        "    _batch_id         VARCHAR(64)  NOT NULL",
        "    _source_system    VARCHAR(64)  NOT NULL DEFAULT 'ERP_E10'",
        "    _source_object    VARCHAR(256) NOT NULL",
        "    _source_row_hash  VARCHAR(64)  NOT NULL",
        "    _ingested_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW()",
        "    _is_deleted       BOOLEAN      NOT NULL DEFAULT FALSE",
    ]

    ddl = f"CREATE TABLE IF NOT EXISTS {raw_table} (\n"
    ddl += ",\n".join(col_defs + tracking_cols)
    ddl += "\n);"
    return ddl
```
