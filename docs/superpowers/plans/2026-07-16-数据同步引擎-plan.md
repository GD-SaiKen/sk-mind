# backend/app/modules/data_sources/service.py — 新增

async def check_connection(source_id: UUID, db: AsyncSession) -> dict:
    """检测数据源连接状态，结果写入 connection_status。"""
    source = await DataSourceDAO(db).get_by_id(source_id)
    if not source:
        raise ValueError(f"DataSource not found: {source_id}")

    is_healthy = False
    try:
        if source.source_category == "database":
            # 暂用占位逻辑：SQL Server 桥接尚未就绪时默认 unknown
            # 后续接入 BridgeAdapter.health_check()
            is_healthy = False  # placeholder
        elif source.source_category == "api":
            import httpx
            fc = source.connector_config or {}
            base_url = fc.get("base_url", "")
            if base_url:
                async with httpx.AsyncClient(timeout=5) as client:
                    resp = await client.get(f"{base_url.rstrip('/')}/health")
                    is_healthy = resp.status_code == 200
    except Exception:
        is_healthy = False

    source.connection_status = "healthy" if is_healthy else "unhealthy"
    source.last_health_check_at = datetime.now(timezone.utc)
    await db.commit()
    return {"source_id": str(source_id), "connection_status": source.connection_status}
```

# backend/app/modules/ingestion/fetchers/bridge_adapter.py
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Any, Optional
import logging

logger = logging.getLogger(__name__)


class BridgeError(Exception):
    """桥接通信异常。"""
    pass


@dataclass
class BridgeCommand:
    source_sql: str
    sync_mode: str
    batch_size: int = 1000
    offset: int = 0
    timestamp_column: Optional[str] = None
    cursor_column: Optional[str] = None
    last_sync_marker: Optional[dict] = None

    @property
    def augmented_sql(self) -> str:
        """根据 sync_mode 在 source_sql 上追加 WHERE/ORDER BY。"""
        sql = self.source_sql.strip()
        if self.sync_mode == "timestamp" and self.timestamp_column:
            marker = (self.last_sync_marker or {}).get("last_timestamp")
            if marker:
                keyword = "AND" if "WHERE" in sql.upper() else "WHERE"
                sql += f" {keyword} {self.timestamp_column} > '{marker}'"
        elif self.sync_mode == "cursor" and self.cursor_column:
            marker = (self.last_sync_marker or {}).get("last_cursor_id")
            if marker is not None:
                keyword = "AND" if "WHERE" in sql.upper() else "WHERE"
                sql += f" {keyword} {self.cursor_column} > {marker}"
                if "ORDER BY" not in sql.upper():
                    sql += f" ORDER BY {self.cursor_column}"
        return sql


class BridgeAdapter(ABC):
    """桥接适配器——与中间服务器通信的抽象接口。"""

    @abstractmethod
    def execute(self, command: BridgeCommand) -> list[dict]:
        ...

    @abstractmethod
    def health_check(self) -> bool:
        ...


class PlaceholderBridgeAdapter(BridgeAdapter):
    """占位桥接适配器 — SQL Server 桥接脚本就绪前的默认实现。

    始终返回空数据，health_check 返回 False。
    """

    def execute(self, command: BridgeCommand) -> list[dict]:
        logger.warning("BridgeAdapter is a placeholder — no real data will be fetched")
        return []

    def health_check(self) -> bool:
        return False
```

# backend/app/modules/ingestion/schema_manager.py

# SQL Server → PostgreSQL 类型映射
SQLSERVER_TYPE_MAP = {
    "nvarchar": "text", "varchar": "text", "text": "text",
    "int": "bigint", "bigint": "bigint", "smallint": "bigint", "tinyint": "bigint",
    "decimal": "numeric", "numeric": "numeric", "money": "numeric",
    "datetime": "timestamp", "datetime2": "timestamp", "smalldatetime": "timestamp",
    "date": "date", "bit": "boolean",
}

TRACKING_COLUMNS = [
    ("_raw_id", "UUID PRIMARY KEY DEFAULT gen_random_uuid()"),
    ("_source_id", "UUID NOT NULL"),
    ("_batch_id", "UUID NOT NULL"),
    ("_pulled_at", "TIMESTAMP NOT NULL"),
    ("_source_signature", "TEXT"),
    ("_row_hash", "TEXT"),
    ("_quality_flags", "JSONB DEFAULT '[]'"),
    ("_ingested_at", "TIMESTAMP DEFAULT now()"),
]


class SchemaManager:
    """Raw schema 下物理表的结构管理器。"""

    def __init__(self, db: Session):
        self._db = db

    def ensure_table(self, schema: str, table: str, columns: list[dict]) -> None:
        """表不存在则 CREATE TABLE。每列: {"name": str, "pg_type": str, "nullable": bool}。"""
        if self._table_exists(schema, table):
            return
        col_defs = [f'    "{c["name"]}" {c["pg_type"]}{"" if c.get("nullable", True) else " NOT NULL"}' for c in columns]
        for name, typedef in TRACKING_COLUMNS:
            col_defs.append(f'    "{name}" {typedef}')
        ddl = f"CREATE TABLE {schema}.{table} (\n" + ",\n".join(col_defs) + "\n)"
        self._db.execute(text(ddl))
        self._db.commit()
        for idx_col in ("_batch_id", "_pulled_at", "_row_hash"):
            self._db.execute(text(f"CREATE INDEX IF NOT EXISTS idx_{table}_{idx_col} ON {schema}.{table} ({idx_col})"))
        self._db.commit()

    def check_schema(self, schema: str, table: str, expected: list[dict]) -> list[SchemaChange]:
        """比对已有表与期望列，返回变更列表。"""
        ...

    def _table_exists(self, schema: str, table: str) -> bool:
        ...
